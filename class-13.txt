);

INSERT INTO student (student_id, student_name, score) VALUES (5, 'Eve', 90);


Question:

calculate the score for students who have the same score

Query:	
	select score,count(score) as student_count from student GROUP BY score;

create table smartphones(
    id INT PRIMARY KEY,
    name VARCHAR(30),
    color VARCHAR(20)
);

INSERT INTO smartphones (id, name, color) VALUES
(1,'Vivo','white');
(2,'Oppo','white');
(3,'Redmi','black');
(4,'Realme','white');
(5,'iPhone','black');
	
Question:
	Calculate the count of each color
Query?
	select color,count(color) as count from smartphones GROUP BY color; 

Output:

	color 	count
	
	white 	3
	black	2
